function toast(msg, duration = 2500) {
  const t = document.getElementById('toast');
  t.textContent = msg;
  t.classList.add('show');
  setTimeout(() => t.classList.remove('show'), duration);

async function api(path, body) {
  const opts = body
    ? { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) }
    : { method: 'GET' };
  const res = await fetch(API + path, opts);
  if (res.status === 401) {
    window.location.href = '/login';
    throw new Error('Session expired');
  }
  if (!res.ok) {
    const err = await res.text();
    throw new Error(err);
  }
  return res.json();
}

}

function hexToRgb(hex) {
  const r = parseInt(hex.slice(1,3), 16);
  const g = parseInt(hex.slice(3,5), 16);
  const b = parseInt(hex.slice(5,7), 16);
  return [r, g, b];
}

function renderCategoryList() {

    chip.dataset.id = cat.id;
    chip.innerHTML = `

function renderCategorySelect() {
  const sel = document.getElementById('saveCategorySelect');
  const prev = sel.value;  // save current selection before rebuild
  sel.innerHTML = '<option value="">— choose category —</option>';
  categories.forEach(cat => {
    const opt = document.createElement('option');
    opt.value = cat.id;
    opt.textContent = cat.name;
    sel.appendChild(opt);
  });
  // Restore previous selection if it's still in the list
  const validPrev = prev && categories.find(c => c.id === prev);
  const validSelected = selectedCategoryId && categories.find(c => c.id === selectedCategoryId);
  if (validPrev) {
    sel.value = prev;
  } else if (validSelected) {
    sel.value = selectedCategoryId;
  } else {
    sel.value = '';
    selectedCategoryId = null;
  }
  onSaveCategoryChange();
}

function selectCategory(id) {
  selectedCategoryId = id;
  // Update sidebar chips
  document.querySelectorAll('.category-chip').forEach(el => {
    el.classList.toggle('selected', el.dataset.id === id);

  // Sync the save dropdown
  const sel = document.getElementById('saveCategorySelect');
  sel.value = id;
  onSaveCategoryChange();
  // Re-render canvas so mask tint updates
  renderCanvas();
}

function onSaveCategoryChange() {
  const sel = document.getElementById('saveCategorySelect');
  selectedCategoryId = sel.value || null;
  const cat = selectedCategoryId ? categories.find(c => c.id === selectedCategoryId) : null;
  // Color the left border based on chosen category
  sel.style.borderLeft = cat ? `3px solid ${cat.color}` : '3px solid #2d333b';

  document.querySelectorAll('.category-chip').forEach(el => {
    el.classList.toggle('selected', el.dataset.id === (selectedCategoryId || ''));
  });
  // Re-render canvas mask tint
  renderCanvas();
}

async function addCategory() {
  const nameEl = document.getElementById('catNameInput');
  const colorEl = document.getElementById('catColorInput');
  const name = nameEl.value.trim();
  if (!name) return toast('Enter a category name');
  if (categories.find(c => c.name.toLowerCase() === name.toLowerCase())) {
    return toast('Category already exists');
  }
  const color = colorEl.value || CAT_PALETTE[categories.length % CAT_PALETTE.length];
  const id = 'cat_' + Date.now() + '_' + Math.random().toString(36).slice(2, 6);
  categories.push({ id, name, color });
  await saveCategories();
  // Auto-select the newly added category

  renderCategoryList();

  sel.value = id;
  onSaveCategoryChange();

}

  renderCategoryList();
  renderCategorySelect();
  renderCanvas();
}

async function browseTo(relPath) {
  try {
    const data = await api('/api/browse', { path: relPath });
    browserCurrentPath = data.current_path;
    document.getElementById('browserPath').textContent = data.full_path;

    const container = document.getElementById('browserContent');
    container.innerHTML = '';

    if (!data.is_root) {
      const parent = document.createElement('div');
      parent.className = 'fb-item dir';
      parent.innerHTML = '<span class="fb-icon">⬆</span><span class="fb-name">..</span>';
      parent.onclick = () => browseTo(data.parent_path);
      container.appendChild(parent);
    }

    data.dirs.forEach(d => {
      const item = document.createElement('div');
      item.className = 'fb-item dir';
      item.innerHTML = `<span class="fb-icon">📁</span><span class="fb-name">${d}</span>`;
      item.onclick = () => {
        const newPath = browserCurrentPath ? browserCurrentPath + '/' + d : d;
        browseTo(newPath);
      };
      container.appendChild(item);
    });

    const videoExts = ['.mp4', '.avi', '.mkv', '.mov', '.webm'];
    data.files.forEach(f => {
      const isVideo = videoExts.includes(f.type);
      const item = document.createElement('div');
      item.className = 'fb-item ' + (isVideo ? 'video' : 'file');
      item.innerHTML = `
        <span class="fb-icon">${isVideo ? '🎬' : '📄'}</span>
        <span class="fb-name">${f.name}</span>
        <span class="fb-size">${f.size_mb} MB</span>
      `;
      if (isVideo) {
        item.onclick = () => {
          const fullPath = data.full_path + '/' + f.name;
          loadVideoFromPath(fullPath, f.name);
          closeFileBrowser();
        };
      }
      container.appendChild(item);
    });

    if (data.dirs.length === 0 && data.files.length === 0) {
      container.innerHTML = '<div style="padding:20px; text-align:center; color:var(--text-dim);">Empty directory</div>';
    }
  } catch (e) {
    toast('Browse error: ' + e.message);
  }
}

async function captureFrame() {
  if (!experiment) return toast('Create an experiment first');
  if (!videoEl.src || videoEl.readyState < 2) return toast('Load a video first');

  const w = videoEl.videoWidth;
  const h = videoEl.videoHeight;
  const offscreen = document.createElement('canvas');
  offscreen.width = w;
  offscreen.height = h;
  const octx = offscreen.getContext('2d');
  octx.drawImage(videoEl, 0, 0, w, h);

  const dataURL = offscreen.toDataURL('image/png');
  const b64 = dataURL.split(',')[1];

  try {
    const data = await api('/api/capture', {
      experiment: experiment,
      frame_b64: b64,
      video_name: currentVideoName || 'unknown',
      timestamp: videoEl.currentTime,
    });

    const img = new Image();
    img.src = dataURL;
    await new Promise(r => { img.onload = r; });

    const frame = {
      id: data.frame_id,
      path: data.frame_path,
      img: img,
      thumb: offscreen.toDataURL('image/jpeg', 0.5),
      timestamp: videoEl.currentTime,
    };
    frames.push(frame);
    addThumb(frames.length - 1);
    selectFrame(frames.length - 1);
    updateCounts();
    toast('Frame captured at ' + videoEl.currentTime.toFixed(2) + 's');
  } catch (e) {
    toast('Capture failed: ' + e.message);
  }
}

// ═══════════════════════════════════════════
// Canvas Rendering
// ═══════════════════════════════════════════
function renderCanvas() {
  if (activeFrameIdx < 0) return;
  const f = frames[activeFrameIdx];
  const img = f.img;
  const area = document.getElementById('canvasArea');
  const aW = area.clientWidth;
  const aH = area.clientHeight;

  const scale = Math.min(aW / img.naturalWidth, aH / img.naturalHeight, 1);
  imgW = Math.floor(img.naturalWidth * scale);
  imgH = Math.floor(img.naturalHeight * scale);
  imgScale = scale;
  imgOffX = Math.floor((aW - imgW) / 2);
  imgOffY = Math.floor((aH - imgH) / 2);

  canvas.width = aW;
  canvas.height = aH;
  ctx.clearRect(0, 0, aW, aH);
  ctx.drawImage(img, imgOffX, imgOffY, imgW, imgH);

  // Draw mask overlay
  if (maskCanvasEl) {
    const opacity = parseFloat(document.getElementById('maskOpacity').value) || 0.4;
    const tmpCanvas = document.createElement('canvas');
    tmpCanvas.width = imgW;
    tmpCanvas.height = imgH;
    const tmpCtx = tmpCanvas.getContext('2d');
    tmpCtx.drawImage(maskCanvasEl, 0, 0, imgW, imgH);
    const maskData = tmpCtx.getImageData(0, 0, imgW, imgH);
    const baseData = ctx.getImageData(imgOffX, imgOffY, imgW, imgH);
    const bd = baseData.data;
    const md = maskData.data;
    const [tR, tG, tB] = activeCategoryRgb();
    for (let i = 0; i < md.length; i += 4) {
      if (md[i] > 128) {
        bd[i]     = Math.round(bd[i]     * (1 - opacity) + tR * opacity);
        bd[i + 1] = Math.round(bd[i + 1] * (1 - opacity) + tG * opacity);
        bd[i + 2] = Math.round(bd[i + 2] * (1 - opacity) + tB * opacity);
      }
    }
    ctx.putImageData(baseData, imgOffX, imgOffY);
  }

  // DINO candidate boxes (dimmed)
  dinoBoxes.forEach((c, i) => {
    if (i === currentBoxIdx) return;
    const [x1, y1, x2, y2] = c.box;
    drawBoxCanvas(x1, y1, x2, y2, 'rgba(210, 153, 34, 0.3)', 1, `box${i + 1}`);
  });

  // Active/editable box
  if (editBox) {
    drawBoxCanvas(editBox.x1, editBox.y1, editBox.x2, editBox.y2, '#3fb950', 2, null);
    drawHandles(editBox);
  } else if (currentBoxIdx >= 0 && dinoBoxes[currentBoxIdx]) {
    const [x1, y1, x2, y2] = dinoBoxes[currentBoxIdx].box;
    drawBoxCanvas(x1, y1, x2, y2, 'var(--box-candidate)', 2, `box${currentBoxIdx + 1}`);
  }

  // Point prompts
  pointPrompts.forEach(pt => {
    const cx = imgOffX + pt.x * imgScale;
    const cy = imgOffY + pt.y * imgScale;
    const isPos = pt.label === 1;
    ctx.beginPath();
    ctx.arc(cx, cy, 6, 0, Math.PI * 2);
    ctx.fillStyle = isPos ? '#3fb950' : '#f85149';
    ctx.fill();
    ctx.strokeStyle = '#fff';
    ctx.lineWidth = 1.5;
    ctx.stroke();
    ctx.fillStyle = '#fff';
    ctx.font = 'bold 9px monospace';
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    ctx.fillText(isPos ? '+' : '−', cx, cy);
    ctx.textAlign = 'start';
    ctx.textBaseline = 'alphabetic';
  });
}

function drawBoxCanvas(x1, y1, x2, y2, color, lineWidth, label) {
  const cx1 = imgOffX + x1 * imgScale;
  const cy1 = imgOffY + y1 * imgScale;
  const cx2 = imgOffX + x2 * imgScale;
  const cy2 = imgOffY + y2 * imgScale;
  ctx.strokeStyle = color;
  ctx.lineWidth = lineWidth;
  ctx.strokeRect(cx1, cy1, cx2 - cx1, cy2 - cy1);
  if (label) {
    ctx.font = '11px monospace';
    ctx.fillStyle = color;
    ctx.fillText(label, cx1 + 3, cy1 - 4);
  }
}

function drawHandles(box) {
  const size = 6;
  const corners = [
    { x: box.x1, y: box.y1 },
    { x: box.x2, y: box.y1 },
    { x: box.x1, y: box.y2 },
    { x: box.x2, y: box.y2 },
  ];
  ctx.fillStyle = '#3fb950';
  corners.forEach(c => {
    const cx = imgOffX + c.x * imgScale;
    const cy = imgOffY + c.y * imgScale;
    ctx.fillRect(cx - size / 2, cy - size / 2, size, size);
  });
}

function setToolMode(mode) {
  toolMode = mode;
  const ids = { box: 'toolBox', pos_point: 'toolPosPoint', neg_point: 'toolNegPoint', brush: 'toolBrush', eraser: 'toolEraser' };
  document.querySelectorAll('.tool-btn').forEach(b => b.classList.remove('active'));
  if (ids[mode]) document.getElementById(ids[mode]).classList.add('active');
  document.getElementById('brushControls').style.display = mode === 'brush' ? 'block' : 'none';
  document.getElementById('eraserControls').style.display = mode === 'eraser' ? 'flex' : 'none';

  // When brush is selected and no mask canvas exists yet, create one
  if (mode === 'brush' && activeFrameIdx >= 0 && !maskCanvasEl) {
    const f = frames[activeFrameIdx];
    ensureMaskCanvas(f.img.naturalWidth, f.img.naturalHeight);
    // Enable approve/save since brush is always an approved edit
    document.getElementById('btnApprove').disabled = false;
    document.getElementById('btnReject').disabled = false;
  }

  renderCanvas();
}

function setBrushMode(mode) {

  document.getElementById('brushModeSub').classList.toggle('active', mode === 'subtract');

function updatePointsList() {

  renderCanvas();
}


// ═══════════════════════════════════════════
// Canvas interactions
// ═══════════════════════════════════════════
canvas.addEventListener('mousedown', e => {
  if (activeFrameIdx < 0) return;
  const { x, y } = canvasToImg(e);

  if (toolMode === 'box') {
    if (!editBox) return;
    const hit = hitTest(x, y, editBox);
    if (hit) {
      dragging = hit;
      dragStart = { x, y, box: { ...editBox } };
      e.preventDefault();
    }
  } else if (toolMode === 'pos_point' || toolMode === 'neg_point') {
    const label = toolMode === 'pos_point' ? 1 : 0;
    pointPrompts.push({ x, y, label });
    updatePointsList();
    renderCanvas();
  } else if (toolMode === 'brush') {
    const f = frames[activeFrameIdx];
    ensureMaskCanvas(f.img.naturalWidth, f.img.naturalHeight);
    isBrushPainting = true;
    paintBrush(x, y);
    renderCanvas();
    e.preventDefault();
  } else if (toolMode === 'eraser') {
    if (!maskCanvasEl) return;
    isErasing = true;
    const r = parseInt(document.getElementById('eraserSize').value) || 20;
    eraseAtPoint(Math.round(x), Math.round(y), r);
    renderCanvas();
    e.preventDefault();
  }
});

canvas.addEventListener('mousemove', e => {
  const { x, y } = canvasToImg(e);

  if (toolMode === 'brush') {
    renderCanvas();
    // Draw brush cursor
    const cx = imgOffX + x * imgScale;
    const cy = imgOffY + y * imgScale;
    const r = parseInt(document.getElementById('brushSize').value) || 12;
    const cr = r * imgScale;
    ctx.beginPath();
    ctx.arc(cx, cy, cr, 0, Math.PI * 2);
    ctx.strokeStyle = brushMode === 'add' ? 'rgba(50,200,255,0.8)' : 'rgba(248,81,73,0.8)';
    ctx.lineWidth = 1.5;
    ctx.stroke();

    if (isBrushPainting && maskCanvasEl) {
      paintBrush(x, y);
      renderCanvas();
      // Redraw cursor after re-render
      const cx2 = imgOffX + x * imgScale;
      const cy2 = imgOffY + y * imgScale;
      ctx.beginPath();
      ctx.arc(cx2, cy2, cr, 0, Math.PI * 2);
      ctx.strokeStyle = brushMode === 'add' ? 'rgba(50,200,255,0.8)' : 'rgba(248,81,73,0.8)';
      ctx.lineWidth = 1.5;
      ctx.stroke();
    }
    return;
  }

  if (toolMode === 'eraser') {
    renderCanvas();
    const cx = imgOffX + x * imgScale;
    const cy = imgOffY + y * imgScale;
    const r = parseInt(document.getElementById('eraserSize').value) || 20;
    const cr = r * imgScale;
    ctx.beginPath();
    ctx.arc(cx, cy, cr, 0, Math.PI * 2);
    ctx.strokeStyle = 'rgba(255,255,255,0.6)';
    ctx.lineWidth = 1;
    ctx.stroke();
    if (isErasing && maskCanvasEl) {
      eraseAtPoint(Math.round(x), Math.round(y), r);
      renderCanvas();
      const cx2 = imgOffX + x * imgScale;
      const cy2 = imgOffY + y * imgScale;
      ctx.beginPath();
      ctx.arc(cx2, cy2, cr, 0, Math.PI * 2);
      ctx.strokeStyle = 'rgba(255,255,255,0.6)';
      ctx.lineWidth = 1;
      ctx.stroke();
    }
    return;
  }

  if (toolMode === 'box') {
    if (!dragging || !editBox) return;
    const dx = x - dragStart.x;
    const dy = y - dragStart.y;
    const ob = dragStart.box;
    if (dragging === 'move') {
      const bw = ob.x2 - ob.x1;
      const bh = ob.y2 - ob.y1;
      editBox.x1 = ob.x1 + dx;
      editBox.y1 = ob.y1 + dy;
      editBox.x2 = editBox.x1 + bw;
      editBox.y2 = editBox.y1 + bh;
    } else if (dragging === 'nw') { editBox.x1 = ob.x1 + dx; editBox.y1 = ob.y1 + dy; }
    else if (dragging === 'ne') { editBox.x2 = ob.x2 + dx; editBox.y1 = ob.y1 + dy; }
    else if (dragging === 'sw') { editBox.x1 = ob.x1 + dx; editBox.y2 = ob.y2 + dy; }
    else if (dragging === 'se') { editBox.x2 = ob.x2 + dx; editBox.y2 = ob.y2 + dy; }
    renderCanvas();
  }
});

canvas.addEventListener('mouseup', () => {
  dragging = null;
  if (isBrushPainting) {
    isBrushPainting = false;
    updateMaskB64();
    // Mark mask as available for approve/save
    maskApproved = false;
    maskScore = 0;
    document.getElementById('btnApprove').disabled = false;
    document.getElementById('btnReject').disabled = false;
    document.getElementById('btnSave').disabled = true;
    document.getElementById('maskInfo').style.display = 'block';
    document.getElementById('maskScoreText').textContent = 'Manual brush mask — click ✓ Approve to save';
  }
  if (isErasing) {
    isErasing = false;
    updateMaskB64();
  }
});

canvas.addEventListener('mouseleave', () => {
  dragging = null;
  if (isBrushPainting) {
    isBrushPainting = false;
    updateMaskB64();
  }
  if (isErasing) {
    isErasing = false;
    updateMaskB64();
  }
  if (activeFrameIdx >= 0) renderCanvas();
});

// ── Brush painting ──
function paintBrush(imgX, imgY) {
  if (!maskCanvasCtx) return;
  const r = parseInt(document.getElementById('brushSize').value) || 12;
  maskCanvasCtx.beginPath();
  maskCanvasCtx.arc(imgX, imgY, r, 0, Math.PI * 2);
  if (brushMode === 'add') {
    maskCanvasCtx.fillStyle = '#ffffff';
    maskCanvasCtx.fill();
  } else {
    // Subtract: erase
    maskCanvasCtx.fillStyle = '#000000';
    maskCanvasCtx.fill();
  }
}

function eraseAtPoint(imgX, imgY, radius) {
  if (!maskCanvasCtx) return;
  maskCanvasCtx.beginPath();
  maskCanvasCtx.arc(imgX, imgY, radius, 0, Math.PI * 2);
  maskCanvasCtx.fillStyle = '#000000';
  maskCanvasCtx.fill();
}

function canvasToImg(e) {
  const rect = canvas.getBoundingClientRect();
  const cx = e.clientX - rect.left;
  const cy = e.clientY - rect.top;
  return {
    x: (cx - imgOffX) / imgScale,
    y: (cy - imgOffY) / imgScale,
  };
}

function hitTest(x, y, box) {
  const tol = 10 / imgScale;
  if (Math.abs(x - box.x1) < tol && Math.abs(y - box.y1) < tol) return 'nw';
  if (Math.abs(x - box.x2) < tol && Math.abs(y - box.y1) < tol) return 'ne';
  if (Math.abs(x - box.x1) < tol && Math.abs(y - box.y2) < tol) return 'sw';
  if (Math.abs(x - box.x2) < tol && Math.abs(y - box.y2) < tol) return 'se';
  if (x >= box.x1 && x <= box.x2 && y >= box.y1 && y <= box.y2) return 'move';
  return null;
}

// DINO
// ═══════════════════════════════════════════
async function runDino() {
  if (activeFrameIdx < 0) return toast('Select a frame first');
  const prompt = document.getElementById('dinoPrompt').value.trim();
  if (!prompt) return toast('Enter a DINO prompt');
  const f = frames[activeFrameIdx];
  const threshold = parseFloat(document.getElementById('dinoThreshold').value);
  document.getElementById('spinDino').classList.add('active');
  document.getElementById('btnRunDino').disabled = true;
  try {
    const data = await api('/api/dino', { frame_path: f.path, prompt, threshold });
    dinoBoxes = data.candidates;
    currentBoxIdx = dinoBoxes.length > 0 ? 0 : -1;
    editBox = null;
    maskImg = null;
    maskCanvasEl = null;
    maskCanvasCtx = null;
    document.getElementById('dotDino').className = 'dot dot-on';
    if (dinoBoxes.length === 0) {
      toast('No detections found. Try lowering the threshold.');
      document.getElementById('boxNav').style.display = 'none';
    } else {
      document.getElementById('boxNav').style.display = 'flex';
      updateBoxNav();
      toast(`Found ${dinoBoxes.length} candidate(s)`);
    }
    document.getElementById('btnSetBox').disabled = dinoBoxes.length === 0;
    renderCanvas();
  } catch (e) {
    toast('DINO error: ' + e.message);
  } finally {
    document.getElementById('spinDino').classList.remove('active');
    document.getElementById('btnRunDino').disabled = false;
  }
}

function prevBox() {
  if (dinoBoxes.length === 0) return;
  currentBoxIdx = (currentBoxIdx - 1 + dinoBoxes.length) % dinoBoxes.length;
  editBox = null; maskImg = null; maskCanvasEl = null; maskCanvasCtx = null;
  updateBoxNav(); renderCanvas();
}

function nextBox() {
  if (dinoBoxes.length === 0) return;
  currentBoxIdx = (currentBoxIdx + 1) % dinoBoxes.length;
  editBox = null; maskImg = null; maskCanvasEl = null; maskCanvasCtx = null;
  updateBoxNav(); renderCanvas();
}

function updateBoxNav() {
  document.getElementById('boxLabel').textContent = `box${currentBoxIdx + 1} / ${dinoBoxes.length}`;
  const c = dinoBoxes[currentBoxIdx];
  document.getElementById('boxScoreArea').innerHTML =
    `<span class="score-badge">${c.label}: ${c.score}</span>`;
}

  document.getElementById('btnRunSam3').disabled = false;

// ═══════════════════════════════════════════
// SAM3
// ═══════════════════════════════════════════
async function runSam3() {
  if (activeFrameIdx < 0) return toast('Select a frame first');
  const f = frames[activeFrameIdx];
  const textPrompt = document.getElementById('sam3Prompt').value.trim();
  const payload = { frame_path: f.path };
  if (editBox) payload.box = [editBox.x1, editBox.y1, editBox.x2, editBox.y2];
  if (textPrompt) payload.text_prompt = textPrompt;
  if (pointPrompts.length > 0) payload.points = pointPrompts.map(p => ({ x: p.x, y: p.y, label: p.label }));
  if (!payload.box && !payload.text_prompt && !payload.points) {
    return toast('Set a box, add points, or enter a text prompt');
  }
  document.getElementById('spinSam3').classList.add('active');
  document.getElementById('btnRunSam3').disabled = true;
  try {
    const data = await api('/api/sam3', payload);
    document.getElementById('dotSam3').className = 'dot dot-on';
    if (data.masks && data.masks.length > 0) {
      const best = data.masks[0];
      maskB64 = best.mask_b64;
      maskScore = best.score;
      maskApproved = false;
      const maskPng = 'data:image/png;base64,' + best.mask_b64;
      maskImg = new Image();
      maskImg.onload = () => {
        maskCanvasEl = document.createElement('canvas');
        maskCanvasEl.width = maskImg.naturalWidth;
        maskCanvasEl.height = maskImg.naturalHeight;
        maskCanvasCtx = maskCanvasEl.getContext('2d');
        maskCanvasCtx.drawImage(maskImg, 0, 0);
        renderCanvas();
      };
      maskImg.src = maskPng;
      document.getElementById('maskInfo').style.display = 'block';
      document.getElementById('maskScoreText').textContent = `Mask score: ${maskScore} — ${data.count} candidate(s)`;
      document.getElementById('btnApprove').disabled = false;
      document.getElementById('btnReject').disabled = false;
      toast('SAM3 mask generated');
    } else {
      toast('No masks returned');
    }
  } catch (e) {
    toast('SAM3 error: ' + e.message);
  } finally {
    document.getElementById('spinSam3').classList.remove('active');
    document.getElementById('btnRunSam3').disabled = false;
  }
}

// Save
// ═══════════════════════════════════════════
async function saveAnnotation() {
  if (!maskApproved || !maskB64) return toast('Approve a mask first');
  if (!experiment) return toast('No experiment set');
  if (!selectedCategoryId) return toast('Choose a category before saving');

  const cat = categories.find(c => c.id === selectedCategoryId);
  if (!cat) return toast('Selected category not found');

  const f = frames[activeFrameIdx];
  const box = editBox
    ? [editBox.x1, editBox.y1, editBox.x2, editBox.y2]
    : (currentBoxIdx >= 0 ? dinoBoxes[currentBoxIdx]?.box : null);
  try {
    const data = await api('/api/save', {
      experiment,
      frame_id: f.id,
      frame_path: f.path,
      box,
      mask_b64: maskB64,
      prompt: document.getElementById('dinoPrompt').value || document.getElementById('sam3Prompt').value,
      label: cat.name,
      category: cat.name,
      category_id: cat.id,
      category_color: cat.color,
      video_name: currentVideoName || 'unknown',
      timestamp: f.timestamp,
      score: maskScore,
    });
    toast('Saved: ' + data.annotation_id + ' [' + cat.name + ']');
    updateCounts();
    loadAnnotations();
  } catch (e) {
    toast('Save error: ' + e.message);
  }
}

    document.getElementById('annViewerCount').textContent = `${annotations.length} annotation(s)`;

    if (annotations.length === 0) {
      document.getElementById('annViewerBody').innerHTML =
        '<div style="padding:40px; text-align:center; color:var(--text-dim);">No annotations saved yet for this experiment.</div>';
      return;
    }

    const body = document.getElementById('annViewerBody');
    body.innerHTML = '';

    for (const ann of annotations) {
      const card = document.createElement('div');
      card.className = 'ann-card';

      // Header
      const boxStr = ann.box ? ann.box.map(v => Math.round(v)).join(', ') : 'N/A';
      const catColor = ann.category_color || '#58a6ff';
      const catName  = ann.category || ann.label || ann.prompt || '—';
      card.innerHTML = `
        <div class="ann-card-header">
          <span class="ann-id">${ann.annotation_id}</span>
          <span style="display:inline-flex;align-items:center;gap:5px;padding:2px 7px;border-radius:3px;background:${catColor}22;border:1px solid ${catColor}55;font-size:11px;font-weight:600;color:${catColor}">
            <span style="width:8px;height:8px;border-radius:50%;background:${catColor};display:inline-block;flex-shrink:0;"></span>
            ${escHtml(catName)}
          </span>
          <span class="ann-meta">score: ${ann.score}</span>
          <span class="ann-meta">t=${ann.timestamp}s</span>
          <span class="ann-meta" style="color:var(--text-dim)">box: [${boxStr}]</span>
          <span class="ann-meta" style="margin-left:auto">${(ann.created_at||'').slice(0,19)}</span>
        </div>
        <div class="ann-card-images">
          <div class="ann-img-cell">
            <div class="cell-label">Frame</div>
            <canvas id="frame-${ann.annotation_id}"></canvas>
          </div>
          <div class="ann-img-cell">
            <div class="cell-label" style="color:${catColor}">Mask · ${escHtml(catName)}</div>
            <canvas id="mask-${ann.annotation_id}"></canvas>
          </div>
          <div class="ann-img-cell">
            <div class="cell-label">Overlay</div>
            <canvas id="overlay-${ann.annotation_id}"></canvas>
          </div>
        </div>`;
      body.appendChild(card);

      // Render images async
      renderAnnCard(ann);
    }
  } catch (e) {
    document.getElementById('annViewerBody').innerHTML =
      `<div style="padding:40px; text-align:center; color:var(--red);">Error loading annotations: ${e.message}</div>`;
  }
}

async function renderAnnCard(ann) {
  // /api/file?path=  serves files relative to EXPERIMENTS_DIR on the backend.
  // ann.frame_path and ann.mask_path are ABSOLUTE paths stored in the JSON,
  // e.g. /sc/arion/.../sam3/experiments/test_mar_17/frames/frame_xxx.png
  //
  // We build the relative path as:  <exp_name>/<subfolder>/<filename>
  // using the experiment name we already have + just the filename from the stored path.
  // This is the same approach as the notebook:
  //   frame_path = exp_dir / "frames" / Path(meta["frame_path"]).name

  const loadImg = (src) => new Promise((res, rej) => {
    const img = new Image();
    // No crossOrigin — same-origin request, setting it can cause CORS failures
    img.onload = () => res(img);
    img.onerror = () => rej(new Error('Failed to load: ' + src));
    img.src = src;
  });

  // Extract just the filename (basename) from an absolute path string
  const basename = (p) => p ? p.split('/').pop() : '';

  const frameRel   = experiment + '/frames/' + basename(ann.frame_path);
  const maskRel    = experiment + '/masks/'  + basename(ann.mask_path);
  const frameUrl   = '/api/file?path=' + encodeURIComponent(frameRel);
  const maskUrl    = '/api/file?path=' + encodeURIComponent(maskRel);

  const markError = (msg) => {
    ['frame', 'mask', 'overlay'].forEach(k => {
      const c = document.getElementById(k + '-' + ann.annotation_id);
      if (!c) return;
      c.width = 320; c.height = 48;
      const cx = c.getContext('2d');
      cx.fillStyle = '#1c2129';
      cx.fillRect(0, 0, 320, 48);
      cx.fillStyle = '#f85149';
      cx.font = '11px monospace';
      cx.fillText('⚠ ' + msg, 8, 28);
    });
  };

  try {
    const [frameImg, maskImg] = await Promise.all([
      loadImg(frameUrl),
      loadImg(maskUrl),
    ]);

    const W = frameImg.naturalWidth;
    const H = frameImg.naturalHeight;

    // Use stored category color, fall back to cyan
    const [tR, tG, tB] = hexToRgb(ann.category_color || '#32c8ff');

    // ── 1. Frame canvas — image + green box overlay ──
    const fc = document.getElementById('frame-' + ann.annotation_id);
    if (!fc) return;
    fc.width = W; fc.height = H;
    const fctx = fc.getContext('2d');
    fctx.drawImage(frameImg, 0, 0);
    if (ann.box) {
      const [bx1, by1, bx2, by2] = ann.box;
      fctx.strokeStyle = ann.category_color || '#3fb950';
      fctx.lineWidth = Math.max(2, Math.round(W / 400));
      fctx.strokeRect(bx1, by1, bx2 - bx1, by2 - by1);
    }

    // ── 2. Read raw mask pixels (grayscale L-mode PNG, R channel = luminance) ──
    const maskTmp = document.createElement('canvas');
    maskTmp.width = W; maskTmp.height = H;
    const maskTmpCtx = maskTmp.getContext('2d');
    maskTmpCtx.drawImage(maskImg, 0, 0, W, H);
    const maskRaw = maskTmpCtx.getImageData(0, 0, W, H).data; // RGBA, R=gray value

    // ── 3. Mask canvas — tinted with category color ──
    const mc = document.getElementById('mask-' + ann.annotation_id);
    if (!mc) return;
    mc.width = W; mc.height = H;
    const mctx = mc.getContext('2d');
    const mData = mctx.createImageData(W, H);
    const md = mData.data;
    for (let i = 0; i < maskRaw.length; i += 4) {
      if (maskRaw[i] > 128) {
        md[i] = tR; md[i+1] = tG; md[i+2] = tB; md[i+3] = 255;
      } else {
        md[i] = 20; md[i+1] = 24; md[i+2] = 31; md[i+3] = 255;
      }
    }
    mctx.putImageData(mData, 0, 0);

    // ── 4. Overlay canvas — frame blended with category color, α=0.45 ──
    const oc = document.getElementById('overlay-' + ann.annotation_id);
    if (!oc) return;
    oc.width = W; oc.height = H;
    const octx = oc.getContext('2d');
    octx.drawImage(frameImg, 0, 0);
    const baseData = octx.getImageData(0, 0, W, H);
    const bd = baseData.data;
    const alpha = 0.45;
    for (let i = 0; i < maskRaw.length; i += 4) {
      if (maskRaw[i] > 128) {
        bd[i]     = Math.round(bd[i]     * (1 - alpha) + tR * alpha);
        bd[i + 1] = Math.round(bd[i + 1] * (1 - alpha) + tG * alpha);
        bd[i + 2] = Math.round(bd[i + 2] * (1 - alpha) + tB * alpha);
      }
    }
    octx.putImageData(baseData, 0, 0);

  } catch (err) {
    markError(err.message);
  }
}

function expandFrame() {
  if (activeFrameIdx < 0) return toast('No frame selected');
  const modal = document.getElementById('expandModal');
  const expCanvas = document.getElementById('expandCanvas');
  const expCtx = expCanvas.getContext('2d');
  const f = frames[activeFrameIdx];
  const img = f.img;
  const maxW = window.innerWidth * 0.95;
  const maxH = window.innerHeight * 0.95;
  const scale = Math.min(maxW / img.naturalWidth, maxH / img.naturalHeight, 1);
  const w = Math.floor(img.naturalWidth * scale);
  const h = Math.floor(img.naturalHeight * scale);
  expCanvas.width = w;
  expCanvas.height = h;
  expCtx.drawImage(img, 0, 0, w, h);

  if (maskCanvasEl) {
    const opacity = parseFloat(document.getElementById('maskOpacity').value) || 0.4;
    const tmpCanvas = document.createElement('canvas');
    tmpCanvas.width = w; tmpCanvas.height = h;
    const tmpCtx = tmpCanvas.getContext('2d');
    tmpCtx.drawImage(maskCanvasEl, 0, 0, w, h);
    const maskData = tmpCtx.getImageData(0, 0, w, h);
    const baseData = expCtx.getImageData(0, 0, w, h);
    const bd = baseData.data; const md = maskData.data;
    const [tR, tG, tB] = activeCategoryRgb();
    for (let i = 0; i < md.length; i += 4) {
      if (md[i] > 128) {
        bd[i]     = Math.round(bd[i]     * (1 - opacity) + tR * opacity);
        bd[i + 1] = Math.round(bd[i + 1] * (1 - opacity) + tG * opacity);
        bd[i + 2] = Math.round(bd[i + 2] * (1 - opacity) + tB * opacity);
      }
    }
    expCtx.putImageData(baseData, 0, 0);
  }

  if (editBox) {
    const s = scale;
    expCtx.strokeStyle = '#3fb950';
    expCtx.lineWidth = 2;
    expCtx.strokeRect(editBox.x1 * s, editBox.y1 * s, (editBox.x2 - editBox.x1) * s, (editBox.y2 - editBox.y1) * s);
  }

  pointPrompts.forEach(pt => {
    expCtx.beginPath();
    expCtx.arc(pt.x * scale, pt.y * scale, 8, 0, Math.PI * 2);
    expCtx.fillStyle = pt.label === 1 ? '#3fb950' : '#f85149';
    expCtx.fill();
    expCtx.strokeStyle = '#fff';
    expCtx.lineWidth = 2;
    expCtx.stroke();
  });

  modal.classList.add('show');
}

function collapseFrame() {
  document.getElementById('expandModal').classList.remove('show');
}
